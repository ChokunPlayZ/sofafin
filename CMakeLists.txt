cmake_minimum_required(VERSION 3.22)
project(JFGamescope VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(Qt6 6.5 REQUIRED COMPONENTS Core Gui Quick QuickControls2 Network OpenGL)
find_package(PkgConfig REQUIRED)
pkg_check_modules(MPV REQUIRED IMPORTED_TARGET mpv>=2.0)
pkg_check_modules(SDL2 REQUIRED IMPORTED_TARGET sdl2>=2.0.14)
qt_standard_project_setup(REQUIRES 6.5)
add_library(jf-core STATIC src/Settings.cpp src/JellyfinClient.cpp src/Playback.cpp)
target_include_directories(jf-core PUBLIC src)
target_link_libraries(jf-core PUBLIC Qt6::Core Qt6::Network)
qt_add_executable(jf-gamescope src/main.cpp src/ControllerInput.cpp src/MpvPlayer.cpp src/MpvVideo.cpp src/AppController.cpp)
target_link_libraries(jf-gamescope PRIVATE jf-core Qt6::Quick Qt6::QuickControls2 Qt6::OpenGL PkgConfig::MPV PkgConfig::SDL2)
qt_add_resources(jf-gamescope ui PREFIX "/" FILES qml/Main.qml qml/NavButton.qml qml/MediaShelf.qml qml/Keyboard.qml assets/logo.svg)
target_compile_options(jf-core PRIVATE -Wall -Wextra -Wpedantic)
target_compile_options(jf-gamescope PRIVATE -Wall -Wextra -Wpedantic)
include(GNUInstallDirs)
install(TARGETS jf-gamescope RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})
install(FILES packaging/io.github.jfgamescope.desktop DESTINATION ${CMAKE_INSTALL_DATADIR}/applications)
install(FILES assets/logo.svg DESTINATION ${CMAKE_INSTALL_DATADIR}/icons/hicolor/scalable/apps RENAME io.github.jfgamescope.svg)
include(CTest)
if(BUILD_TESTING)
    find_package(Qt6 REQUIRED COMPONENTS Test)
    qt_add_executable(jf-tests tests/CoreTests.cpp)
    target_link_libraries(jf-tests PRIVATE jf-core Qt6::Test)
    qt_add_executable(jf-live-check tests/LiveCheck.cpp)
    target_link_libraries(jf-live-check PRIVATE jf-core)
    add_test(NAME core COMMAND jf-tests)
    add_test(NAME ui-smoke COMMAND jf-gamescope --windowed --smoke-test)
    set_tests_properties(ui-smoke PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen;QT_QUICK_BACKEND=software;XDG_CONFIG_HOME=${CMAKE_BINARY_DIR}/test-config" TIMEOUT 15)
endif()
