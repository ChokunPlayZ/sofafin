cmake_minimum_required(VERSION 3.22)
project(Sofafin VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(Qt6 6.5 REQUIRED COMPONENTS Core Gui Quick QuickControls2 Network OpenGL DBus)
find_package(PkgConfig REQUIRED)
pkg_check_modules(MPV REQUIRED IMPORTED_TARGET mpv>=2.0)
pkg_check_modules(SDL2 REQUIRED IMPORTED_TARGET sdl2>=2.0.14)
qt_standard_project_setup(REQUIRES 6.5)
add_library(sofafin-core STATIC src/Settings.cpp src/JellyfinClient.cpp src/Playback.cpp)
target_include_directories(sofafin-core PUBLIC src)
target_link_libraries(sofafin-core PUBLIC Qt6::Core Qt6::Network)
add_library(sofafin-native STATIC src/ControllerInput.cpp src/MpvPlayer.cpp src/MpvVideo.cpp src/AppController.cpp src/BlurHash.cpp src/ScreenSaverInhibitor.cpp)
target_link_libraries(sofafin-native PUBLIC sofafin-core Qt6::Quick Qt6::QuickControls2 Qt6::OpenGL PkgConfig::MPV PkgConfig::SDL2)
target_link_libraries(sofafin-native PRIVATE Qt6::DBus)
qt_add_executable(sofafin src/main.cpp)
target_link_libraries(sofafin PRIVATE sofafin-native)
set(SOFAFIN_UI qml/Main.qml qml/NavButton.qml qml/IconButton.qml qml/ProfileButton.qml qml/MediaShelf.qml qml/MediaGrid.qml qml/MediaCard.qml qml/Artwork.qml qml/DetailView.qml qml/PlaybackOptions.qml qml/OptionSelector.qml qml/PlayerView.qml qml/SeekBar.qml qml/SkeletonBlock.qml qml/SkeletonShelf.qml assets/logo.svg assets/search.svg assets/settings.svg assets/close.svg assets/back.svg)
qt_add_resources(sofafin ui PREFIX "/" FILES ${SOFAFIN_UI})
target_compile_options(sofafin-core PRIVATE -Wall -Wextra -Wpedantic)
target_compile_options(sofafin PRIVATE -Wall -Wextra -Wpedantic)
target_compile_options(sofafin-native PRIVATE -Wall -Wextra -Wpedantic)
include(GNUInstallDirs)
install(TARGETS sofafin RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})
install(FILES packaging/io.github.sofafin.desktop DESTINATION ${CMAKE_INSTALL_DATADIR}/applications)
install(FILES assets/logo.svg DESTINATION ${CMAKE_INSTALL_DATADIR}/icons/hicolor/scalable/apps RENAME io.github.sofafin.svg)
include(CTest)
if(BUILD_TESTING)
    find_package(Qt6 REQUIRED COMPONENTS Test)
    qt_add_executable(sofafin-tests tests/CoreTests.cpp)
    target_link_libraries(sofafin-tests PRIVATE sofafin-core Qt6::Test)
    qt_add_executable(sofafin-live-check tests/LiveCheck.cpp)
    target_link_libraries(sofafin-live-check PRIVATE sofafin-core)
    qt_add_executable(sofafin-ui-tests tests/NativeTests.cpp)
    target_link_libraries(sofafin-ui-tests PRIVATE sofafin-native Qt6::Test)
    qt_add_resources(sofafin-ui-tests test_ui PREFIX "/" FILES ${SOFAFIN_UI})
    qt_add_executable(sofafin-live-ui-tests tests/LiveUiTests.cpp)
    target_link_libraries(sofafin-live-ui-tests PRIVATE sofafin-native Qt6::Test)
    qt_add_resources(sofafin-live-ui-tests live_ui PREFIX "/" FILES ${SOFAFIN_UI})
    add_test(NAME native COMMAND sofafin-ui-tests)
    set_tests_properties(native PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen;QT_QUICK_BACKEND=software" TIMEOUT 90)
    add_test(NAME core COMMAND sofafin-tests)
    if(CMAKE_SYSTEM_NAME STREQUAL "Linux")
        find_program(DBUS_RUN_SESSION_EXECUTABLE dbus-run-session REQUIRED)
        qt_add_executable(sofafin-screensaver-tests tests/ScreenSaverTests.cpp)
        target_link_libraries(sofafin-screensaver-tests PRIVATE sofafin-native Qt6::Test Qt6::DBus)
        add_test(NAME screensaver COMMAND ${DBUS_RUN_SESSION_EXECUTABLE} -- $<TARGET_FILE:sofafin-screensaver-tests>)
        set_tests_properties(screensaver PROPERTIES TIMEOUT 30)
        pkg_check_modules(AVUTIL REQUIRED IMPORTED_TARGET libavutil)
        add_library(sofafin-hwdec-guard SHARED tests/HardwareDecodeGuard.cpp)
        target_link_libraries(sofafin-hwdec-guard PRIVATE PkgConfig::AVUTIL)
        qt_add_executable(sofafin-playback-startup-tests tests/PlaybackStartupTests.cpp)
        target_link_libraries(sofafin-playback-startup-tests PRIVATE sofafin-native Qt6::Test PkgConfig::AVUTIL)
        target_compile_definitions(sofafin-playback-startup-tests PRIVATE SOFAFIN_HWDEC_GUARD_PATH="$<TARGET_FILE:sofafin-hwdec-guard>")
        add_dependencies(sofafin-playback-startup-tests sofafin-hwdec-guard)
        add_test(NAME playback-startup COMMAND sofafin-playback-startup-tests)
        set_tests_properties(playback-startup PROPERTIES TIMEOUT 40)
    endif()
    add_test(NAME ui-smoke COMMAND sofafin --windowed --smoke-test)
    set_tests_properties(ui-smoke PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen;QT_QUICK_BACKEND=software;XDG_CONFIG_HOME=${CMAKE_BINARY_DIR}/test-config" TIMEOUT 15)
endif()
